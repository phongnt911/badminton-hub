const privateProjectionTokens = <String>[
  'contact',
  'phone',
  'bank',
  'account',
  'receiver',
];

Set<String> _normalizedKeys(Map value) =>
    value.keys.map((key) => key.toString().toLowerCase()).toSet();

bool _hasPrivateProjectionField(Object? value) {
  if (value is Map) {
    for (final entry in value.entries) {
      final key = entry.key.toString().toLowerCase();
      if (privateProjectionTokens.any(key.contains)) return true;
      if (_hasPrivateProjectionField(entry.value)) return true;
    }
  } else if (value is List) {
    for (final item in value) {
      if (_hasPrivateProjectionField(item)) return true;
    }
  }
  return false;
}

List<String> _projectionIssues(Object? value, Set<String> allowedKeys) {
  if (value is! Map) return const ['payload is not an object'];
  final keys = _normalizedKeys(value);
  final issues = <String>[];
  final missing = allowedKeys.difference(keys).toList()..sort();
  final extra = keys.difference(allowedKeys).toList()..sort();
  if (missing.isNotEmpty) issues.add('missing=${missing.join(',')}');
  if (extra.isNotEmpty) issues.add('extra=${extra.join(',')}');
  if (_hasPrivateProjectionField(value)) issues.add('private-key');
  return issues;
}

List<String> _historyProjectionIssues(Object? value) {
  if (value is! List) return const ['payload is not an array'];
  final issues = <String>[];
  if (value.isEmpty) issues.add('empty-history');
  for (var index = 0; index < value.length; index++) {
    final rowIssues = _projectionIssues(value[index], historyProjectionKeys);
    if (rowIssues.isNotEmpty) {
      issues.add('row[$index]: ${rowIssues.join(';')}');
    }
  }
  return issues;
}

void main() {
  Map<String,dynamic> history() => {for(final key in historyProjectionKeys) key:null};
  Map<String,dynamic> summary() => {for(final key in summaryProjectionKeys) key:null};
  test('review control: valid whitelist payload remains accepted', () {
    expect(_projectionIssues(summary(),summaryProjectionKeys),isEmpty);
    expect(_historyProjectionIssues([history(),history()]),isEmpty);
  });
  test('review H2: original bank_account counterexample is rejected', () {
    final row=history()..['bank_account']='REVIEW-SENTINEL';
    expect(_historyProjectionIssues([row]),isNotEmpty);
  });
  test('review H2: original phone on later row is rejected', () {
    final row=history()..['phone']='REVIEW-SENTINEL';
    expect(_historyProjectionIssues([history(),row]).any((issue)=>issue.contains('row[1]')),isTrue);
  });
  test('review H2: receiver snapshot and private nested under allowed key rejected', () {
    final a=summary()..['receiver_snapshot']={'account_number':'REVIEW-SENTINEL'};
    final b=summary()..['club_name']={'safe_label':'A','receiver_snapshot':{'account_number':'REVIEW-SENTINEL'}};
    expect(_projectionIssues(a,summaryProjectionKeys),isNotEmpty);
    expect(_projectionIssues(b,summaryProjectionKeys),contains('private-key'));
  });
}
